ssh -i "C:\Users\kumag\.ssh\kagoya\kagoya.key" ubuntu@133.18.123.23

▼DB操作
1. 管理者（postgresユーザー）に切り替える
sudo -i -u postgres

2. PostgreSQLの対話モード（psql）を起動する
psql

-- 1. ユーザー（ロール）の作成（'myuser' と 'mypassword' は好きなものに変えてください）
CREATE USER ippcdb WITH PASSWORD '6@B4m2Vt';

-- 2. データベースの作成
CREATE DATABASE db_kintai;

-- 3. データベースの所有権をユーザーに付与
GRANT ALL PRIVILEGES ON DATABASE db_kintai TO ippcdb;

3. DBのトンネリング(PowerShellで実施)
ssh -L 5432:127.0.0.1:5432 ubuntu@133.18.123.23 -i "C:\Users\kumag\.ssh\kagoya\kagoya.key" -N
ssh -v -L 5432:127.0.0.1:5432 ubuntu@133.18.123.23 -i "C:\Users\kumag\.ssh\kagoya\kagoya.key" -N
<!-- -vでログ出す -->

4. 権限不足にならないよう権限付与
psql -U postgres -d db_kintai -c "GRANT ALL ON SCHEMA public TO ippcdb;"

▼プロダクションデプロイ手順
【初回】
1. サーバーログイン
   ssh -i "C:\Users\kumag\.ssh\kagoya\kagoya.key" ubuntu@133.18.123.23

2. リポジトリ取得
   cd ~
   git clone <リポジトリURL> kintai
   cd kintai

3. .env.local 作成
   nano .env.local
   ---
   DATABASE_URL=postgresql://ippcdb:<パスワード>@localhost:5432/db_kintai
   AUTH_SECRET=<openssl rand -base64 32 で生成>
   AUTH_GOOGLE_ID=<Google Cloud ConsoleのクライアントID>
   AUTH_GOOGLE_SECRET=<Google Cloud Consoleのシークレット>
   AUTH_URL=https://kintai.iwaki-i.online
   ---

4. ビルド＆起動
   npm ci
   npx prisma generate
   npx prisma migrate deploy
   npm run build
   pm2 start npm --name kintai -- start
   pm2 save
   pm2 startup   ← 表示されたコマンドを実行

【更新デプロイ（2回目以降）】
   cd ~/kintai
   git pull
   npm ci                      ← package変更があれば
   npx prisma generate         ← スキーマ変更があれば
   npx prisma migrate deploy   ← スキーマ変更があれば
   npm run build
   pm2 restart kintai

【確認コマンド】
   pm2 status
   pm2 logs kintai --lines 50

【Google OAuth リダイレクトURI】
   https://kintai.iwaki-i.online/api/auth/callback/google
   ※ Google Cloud Console → OAuth 2.0 クライアント → 承認済みリダイレクト URI に追加
